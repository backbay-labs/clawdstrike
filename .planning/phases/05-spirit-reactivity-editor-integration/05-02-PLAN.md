Token tinting strategy per spirit kind (subtle ±15% hue/saturation shift):

- oracle: propertyName + keyword → blend #d4a84b toward #7b68ee = ~#a080b8; string → blend #3dbf84 toward #7b68ee = ~#5d76c9
- witness: no change (accent already matches defaults)

- default/null: keep existing token colors

Simple blendHex inline helper (avoid importing from scene-math.ts — yaml-editor has no R3F deps):
```typescript
function blendHex(base: string, target: string, t: number): string {
  const parse = (h: string) => [
    parseInt(h.slice(1,3),16), parseInt(h.slice(3,5),16), parseInt(h.slice(5,7),16)
  ];
  const [br,bg,bb] = parse(base); const [tr,tg,tb] = parse(target);
  const r = Math.round(br + (tr-br)*t).toString(16).padStart(2,'0');
  const g = Math.round(bg + (tg-bg)*t).toString(16).padStart(2,'0');
  const b = Math.round(bb + (tb-bb)*t).toString(16).padStart(2,'0');
  return `#${r}${g}${b}`;
}
```

function createHighlightStyle(accentColor?: string | null): ReturnType<typeof HighlightStyle.define> {
  // Determine per-kind token overrides
  const acc = accentColor ?? "#d4a84b";
  // For each spirit kind, blend specific tags toward accent at t=0.35
  // oracle (#7b68ee): propertyName + keyword blend gold → purple
  // specter (#c45c5c): operator + punctuation blend steel → red
  // sentinel/witness: no meaningful shift from defaults
  const propColor = blendHex("#d4a84b", acc, 0.35);   // property names
  const strColor  = blendHex("#3dbf84", acc, 0.25);   // strings (subtle)
  const opColor   = blendHex("#6f7f9a", acc, 0.35);   // operators

  return HighlightStyle.define([
    { tag: tags.propertyName, color: propColor },
    { tag: tags.definition(tags.propertyName), color: propColor },
    { tag: tags.string, color: strColor },
    { tag: tags.number, color: "#6f7f9a" },
    { tag: tags.integer, color: "#6f7f9a" },
    { tag: tags.float, color: "#6f7f9a" },

}
