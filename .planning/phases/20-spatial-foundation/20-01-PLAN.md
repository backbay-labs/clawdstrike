    **Step 5: WebGPU renderer swap in Canvas gl prop**
    In `ObservatoryWorldCanvas.tsx`, change the `<Canvas>` `gl` prop from:

    to an async function that tries WebGPU first:
    ```tsx
    gl={async (canvas) => {
      try {
        const { WebGPURenderer } = await import('three/webgpu');
        const renderer = new WebGPURenderer({ canvas, antialias: true, logarithmicDepthBuffer: true });
        await renderer.init();
        return renderer;
      } catch {
        // Fallback to WebGL2
        return new THREE.WebGLRenderer({
          canvas,
          antialias: false,
          alpha: false,
          powerPreference: "high-performance",
          logarithmicDepthBuffer: true,
        });
      }
    }}
    ```
