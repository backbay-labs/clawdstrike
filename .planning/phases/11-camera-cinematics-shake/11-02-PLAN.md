At the top of the useFrame callback, BEFORE the normal `if (!initializedRef.current)` block, add the fly-by branch:

```typescript
// CAM-01: fly-by sequencing — runs before normal tracking when flyByActive=true
if (flyByActive && !flyByCompleteCalledRef.current) {
  // On first frame of fly-by: initialize position and launch leg 0
  if (!initializedRef.current) {
    const wp = FLY_BY_WAYPOINTS[0];
    controls.object.position.set(...wp.position as [number, number, number]);
    controls.target.set(...wp.target as [number, number, number]);
    controls.update();
    initializedRef.current = true;
    // Don't launch flight on first frame — skip to next waypoint immediately
    waypointIndexRef.current = 0;
    return;
  }
  // Launch flight to current waypoint if no flight is in progress
  if (!flightRef.current) {
    const idx = waypointIndexRef.current;
    if (idx >= FLY_BY_WAYPOINTS.length) {
      // All waypoints done
      flyByCompleteCalledRef.current = true;
      onFlyByComplete();
      return;
    }
    const wp = FLY_BY_WAYPOINTS[idx];
    const from = controls.object.position.clone();
    const to = new THREE.Vector3(...wp.position as [number, number, number]);
    const toTarget = new THREE.Vector3(...wp.target as [number, number, number]);
    const fromTarget = controls.target.clone();
    const axis = to.clone().sub(from);
    const travelDist = axis.length();
    // Guard: skip near-zero travel legs
    if (travelDist < 0.5) {
      waypointIndexRef.current = idx + 1;
      return;
    }
    const lateral = new THREE.Vector3(-axis.z, 0, axis.x)
      .normalize()
      .multiplyScalar(Math.min(2.4, 0.8 + travelDist * 0.04));
    const via = from.clone().lerp(to, 0.5).add(lateral).setY(Math.max(from.y, to.y) + camera.arrivalLift + 1.2);
    const viaTarget = fromTarget.clone().lerp(toTarget, 0.5).setY(Math.max(fromTarget.y, toTarget.y) + camera.arrivalLift * 0.32);
    flightRef.current = {
      startTime: clock.elapsedTime,
      duration: wp.durationMs / 1000,
      fromPosition: from,
      viaPosition: via,
      toPosition: to,
      fromTarget,
      viaTarget,
      toTarget,
    };
  }
  // Run flight (reuse existing flight execution block verbatim)
  if (flightRef.current) {
    const progress = (clock.elapsedTime - flightRef.current.startTime) / flightRef.current.duration;
    if (progress >= 1) {
      controls.object.position.copy(flightRef.current.toPosition);
      controls.target.copy(flightRef.current.toTarget);
      controls.update();
      flightRef.current = null;
      waypointIndexRef.current += 1;
      return;
    }
    const eased = smoothstep01(progress);
    const travelPos = bezierPoint(flightRef.current.fromPosition, flightRef.current.viaPosition, flightRef.current.toPosition, eased);
    const travelTgt = bezierPoint(flightRef.current.fromTarget, flightRef.current.viaTarget, flightRef.current.toTarget, eased);
    controls.object.position.copy(travelPos);
    controls.target.copy(travelTgt);
    controls.update();
    return;
  }
  return;
}
// When flyByActive becomes false (hand-off), reset fly-by state for potential future use
if (!flyByActive && flyByCompleteCalledRef.current) {
  // Nothing to reset — flyByCompleteCalledRef stays true to prevent re-trigger
}
```
