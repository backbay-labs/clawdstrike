**Complete component shape:**
```tsx
export function GhostTraceLayer({ traces, opacityScale }: GhostTraceLayerProps) {
  const byStation = useMemo(/* group by stationId */, [traces]);
  if (byStation.size === 0) return null;
  return (
    <>
      {Array.from(byStation.entries()).map(([stationId, stationTraces]) => {
        const pos = OBSERVATORY_STATION_POSITIONS[stationId];
        if (!pos) return null;
        return (
          <GhostStationMarkers
            key={stationId}
            position={[pos[0], pos[1] + 1.2, pos[2]]}
            traces={stationTraces}
            opacityScale={opacityScale}
          />
        );
      })}
    </>
  );
}
```
