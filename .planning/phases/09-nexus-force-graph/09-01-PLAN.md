interface NexusState {
  strikecells: Strikecell[];
  connections: StrikecellConnection[];
  actions: {
    setStrikecells: (cells: Strikecell[]) => void;
    setConnections: (conns: StrikecellConnection[]) => void;
  };
}

const useNexusStoreBase = create<NexusState>((set) => ({
  strikecells: DEMO_STRIKECELLS,
  connections: DEMO_CONNECTIONS,
  actions: {
    setStrikecells: (cells) => set({ strikecells: cells }),
    setConnections: (conns) => set({ connections: conns }),
  },
}));

export const useNexusStore = createSelectors(useNexusStoreBase);

```typescript
// NexusForceCanvas — standalone R3F Canvas with force-directed graph layout.
// Uses r3f-forcegraph (vasturiano) which depends only on three >=0.154 (no R3F peer dep).
// Peer dep verified: npm info r3f-forcegraph peerDependencies = { react: '*', three: '>=0.154' }
// Workbench has three ^0.170.0 — compatible.

import { useRef, useCallback, useMemo } from "react";
import { Canvas } from "@react-three/fiber";
import { OrbitControls } from "@react-three/drei";
import ForceGraph from "r3f-forcegraph";
import { useNexusStore } from "../stores/nexus-store";
import { usePaneStore } from "@/features/panes/pane-store";
import { STRIKECELL_ROUTE_MAP } from "../types";
import type { StrikecellDomainId } from "../types";

// Color palette by domain — distinct hues for visual separation
const DOMAIN_COLOR: Record<StrikecellDomainId, string> = {
  "security-overview": "#3dbf84",   // sentinel green
  "threat-radar":      "#c45c5c",   // specter red
  "attack-graph":      "#c45c5c",   // specter red (related)
  "network-map":       "#7b68ee",   // oracle purple
  "workflows":         "#7b68ee",   // oracle purple (related)
  "marketplace":       "#d4a84b",   // witness gold
  "events":            "#d4a84b",   // witness gold (related)
  "policies":          "#3dbf84",   // sentinel green (related)
  "forensics-river":   "#c45c5c",   // specter red (related)
};

interface GraphNode {
  id: string;
  name: string;
  color: string;
  val: number; // node size weight
}

interface GraphLink {
  source: string;
  target: string;
  value: number; // link width weight
}

interface GraphData {
  nodes: GraphNode[];
  links: GraphLink[];
}

export function NexusForceCanvas() {
  const strikecells = useNexusStore.use.strikecells();
  const connections = useNexusStore.use.connections();
  // eslint-disable-next-line @typescript-eslint/no-explicit-any
  const graphRef = useRef<any>(null);

  const graphData: GraphData = useMemo(() => ({
    nodes: strikecells.map((sc) => ({
      id: sc.id,
      name: sc.name,
      color: DOMAIN_COLOR[sc.id] ?? "#888888",
      val: 1.5,
    })),
    links: connections.map((c) => ({
      source: c.sourceId,
      target: c.targetId,
      value: c.strength,
    })),
  }), [strikecells, connections]);

  const handleNodeClick = useCallback((node: GraphNode) => {
    const route = STRIKECELL_ROUTE_MAP[node.id as StrikecellDomainId];
    if (route) {
      usePaneStore.getState().openApp(route);
    }
  }, []);

  const handleNodeDrag = useCallback((node: GraphNode & { fx?: number; fy?: number; fz?: number }, translate: { x: number; y: number; z: number }) => {
    node.fx = node.x !== undefined ? (node as unknown as { x: number }).x + translate.x : translate.x;
    node.fy = node.y !== undefined ? (node as unknown as { y: number }).y + translate.y : translate.y;
    node.fz = node.z !== undefined ? (node as unknown as { z: number }).z + translate.z : translate.z;
  }, []);

  const handleNodeDragEnd = useCallback((node: GraphNode & { fx?: number; fy?: number; fz?: number }) => {
    // Release pin so simulation can continue
    node.fx = undefined;
    node.fy = undefined;
    node.fz = undefined;
    graphRef.current?.d3ReheatSimulation();
  }, []);

  return (
    <div className="relative flex-1 overflow-hidden" data-testid="nexus-force-canvas">
      <div className="absolute inset-0">
        <Canvas
          camera={{ position: [0, 0, 200], fov: 60 }}
          frameloop="demand"
          dpr={[1, 1.8]}
        >
          <ambientLight intensity={0.6} />
          <pointLight position={[100, 100, 100]} intensity={0.8} />
          <OrbitControls makeDefault enablePan enableZoom enableRotate />
          <ForceGraph
            ref={graphRef}
            graphData={graphData}
            nodeAutoColorBy="color"
            nodeColor={(node: GraphNode) => node.color}
            nodeLabel={(node: GraphNode) => node.name}
            nodeVal={(node: GraphNode) => node.val}
            linkColor={() => "rgba(255,255,255,0.2)"}
            linkWidth={(link: GraphLink) => link.value * 2}
            onNodeClick={handleNodeClick}
            onNodeDrag={handleNodeDrag}
            onNodeDragEnd={handleNodeDragEnd}
            d3AlphaDecay={0.02}
            d3VelocityDecay={0.3}
            cooldownTicks={150}
          />
        </Canvas>
      </div>
    </div>
  );
}
```

**Note on r3f-forcegraph API:** The default export is `ForceGraph` (or named `R3fForceGraph` in some versions). Check the actual export name after install: `import ForceGraph from 'r3f-forcegraph'` uses the default export. If the package exports `{ ForceGraph3D }` as named, adjust accordingly. Look at `node_modules/r3f-forcegraph/dist/r3f-forcegraph.js` or the README.

**Note on node position in drag handler:** The `r3f-forcegraph` drag callback signature for `onNodeDrag` is `(node, translate)` where translate is `{x, y, z}` delta. Pin by setting `node.fx = node.x + translate.x`. The actual `node.x/y/z` are mutable properties on the d3-force node object.

**TypeScript:** r3f-forcegraph may not have perfect TS types. Use `// eslint-disable-next-line @typescript-eslint/no-explicit-any` for `graphRef` and cast node types as needed. The component will still be type-safe at its prop boundary.
