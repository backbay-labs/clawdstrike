**Note on NexusLayoutMode:** The type union is `"radial" | "typed-lanes" | "force-directed"`. The current NexusTab hardcodes `mode="atlas"` on ObservatoryWorldCanvas, but `"atlas"` is not in NexusLayoutMode. The default store value should be `"radial"` — this is the non-force mode (ObservatoryWorldCanvas). NexusTab will render ObservatoryWorldCanvas when `layoutMode !== "force-directed"` and NexusForceCanvas when `layoutMode === "force-directed"`.

Add `layoutMode` to the store:

```typescript
import { create } from "zustand";
import { createSelectors } from "@/lib/create-selectors";
import type { Strikecell, StrikecellConnection, NexusLayoutMode } from "../types";
import { DEMO_STRIKECELLS, DEMO_CONNECTIONS } from "../types";

interface NexusState {
  strikecells: Strikecell[];
  connections: StrikecellConnection[];
  layoutMode: NexusLayoutMode;
  actions: {
    setStrikecells: (cells: Strikecell[]) => void;
    setConnections: (conns: StrikecellConnection[]) => void;
    setLayoutMode: (mode: NexusLayoutMode) => void;
  };
}

const useNexusStoreBase = create<NexusState>((set) => ({
  strikecells: DEMO_STRIKECELLS,
  connections: DEMO_CONNECTIONS,
  layoutMode: "radial",
  actions: {
    setStrikecells: (cells) => set({ strikecells: cells }),
    setConnections: (conns) => set({ connections: conns }),
    setLayoutMode: (mode) => set({ layoutMode: mode }),
  },
}));

export const useNexusStore = createSelectors(useNexusStoreBase);
```
